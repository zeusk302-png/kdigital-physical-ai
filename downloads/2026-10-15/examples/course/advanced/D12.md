# D12 · CSV 검사에서 JSON 보고서까지 연결하기

## 무엇을 하는 개념인가요?

파일 읽기·값 검사·집계·저장을 각각 배웠다면 이제 하나의 작업으로 연결합니다. 원본은 유지하고 사용할 기록과 오류를 나누어 결과 파일을 만듭니다. 완성 프로그램은 짧은 기능들을 함수로 나누어 조립합니다.

**앞에서 확인할 것:** D07 CSV·D08 JSON·D09 검사·D11 모듈를 먼저 확인하세요.

## 구성요소와 기호

| 요소 | 뜻과 사용 위치 |
|---|---|
| 읽기 → 검사 → 집계 → 저장 | 각 단계의 함수가 어떤 값을 받아 무엇을 반환하는지 먼저 정합니다. |
| 열 구조 검사 | 필수 열과 열 개수를 확인해야 잘못된 표를 정상 자료로 오해하지 않습니다. |
| 중복과 변환 | 번호·숫자·참거짓을 각각 정한 규칙으로 검사합니다. |
| 결과 보고서 | 입력 수·채택 수·미측정 수·평균·오류 사유를 함께 저장합니다. |
| 재읽기와 검산 | 저장한 JSON을 다시 읽고 화면 요약과 일치하는지 확인합니다. |
| 원본 보존 | data는 읽기 전용으로 다루고 결과는 work에 저장합니다. |

괄호 `()`는 호출할 값을 묶고, 콜론 `:` 다음 들여쓴 줄들은 함수·조건·반복의 몸통입니다. 대괄호 `[]`는 목록 또는 위치·키로 값 찾기, 중괄호 `{}`는 딕셔너리 또는 f-string 안 값을 넣을 자리에 사용합니다. 같은 기호도 사용 위치에 따라 의미가 다릅니다.

## 실행 과정

첫 예제는 CSV 7행을 읽어 4행을 채택하고 3행의 문제를 분리합니다. 채택된 4행 중 하나는 미측정이므로 평균에는 측정 3개만 사용합니다. 경고 목록과 오류 목록을 JSON에 함께 저장합니다.

```python
# D12 · CSV 검수부터 JSON 보고서까지
# 파일 전체를 복사해 UTF-8로 저장합니다. # 뒤의 글은 실행하지 않는 주석입니다.
# 배울 내용: 함수, CSV, 자료형, 오류 목록, JSON 저장
# 실행: course 폴더에서 python -X utf8 advanced/D12_01_normal.py
# print(...)의 괄호는 출력할 값을 묶고, 문자열 양끝 따옴표는 글자의 범위를 나타냅니다.
# 먼저 입력값을 보고 결과를 예상한 다음, 실제 출력과 비교하세요.

import csv
import json
import math
from pathlib import Path
from sample_helpers import normalize_active, summarize_values

def load_records(path):
    # 파일 읽기와 업무 검수를 다른 함수로 나눕니다.
    with path.open(encoding="utf-8-sig", newline="") as file:
        reader = csv.DictReader(file)
        required = {"sensor_id", "temperature_c", "active"}
        if not required.issubset(reader.fieldnames or []):
            raise ValueError("sensor_id, temperature_c, active 열이 필요합니다")
        return list(reader)

def validate_rows(rows):
    accepted = []
    issues = []
    seen = set()
    for line_number, row in enumerate(rows, start=2):
        # CSV 첫 줄은 열 이름이므로 데이터 줄 번호는 2부터 시작합니다.
        # 열이 넘치면 None 키, 부족하면 None 값이 생깁니다. 빈 문자열과 다릅니다.
        wrong_shape = None in row
        for value in row.values():
            if value is None:
                wrong_shape = True
        if wrong_shape:
            issues.append({"line": line_number, "reason": "열 개수 오류"})
            continue
        sensor_id = row["sensor_id"].strip()
        if sensor_id == "":
            issues.append({"line": line_number, "reason": "번호 누락"})
            continue
        # 원본에 번호가 반복되면 뒤쪽 기록을 임의로 덮어쓰지 않습니다.
        if sensor_id in seen:
            issues.append({"line": line_number, "reason": "번호 중복"})
            continue
        seen.add(sensor_id)
        raw = row["temperature_c"].strip()
        try:
            temperature = None if raw == "" else float(raw)
            # float('NaN')은 변환에 성공하지만 정상 측정값으로 쓰면 안 됩니다.
            if temperature is not None and not math.isfinite(temperature):
                raise ValueError("유한한 온도가 필요합니다")
            active = normalize_active(row["active"])
        except ValueError:
            # 문제 원문과 줄 번호를 남겨 담당자가 다시 확인할 수 있게 합니다.
            issues.append({"line": line_number, "reason": "온도 또는 상태 오류", "raw": row})
            continue
        accepted.append({"id": sensor_id, "temperature": temperature, "active": active})
    return accepted, issues

def make_report(records, issues, input_count):
    values = []
    alerts = []
    for record in records:
        temperature = record["temperature"]
        values.append(temperature)
        # 교육용 기준입니다. 실제 장비 제어 명령을 보내지 않습니다.
        if record["active"] and temperature is not None and temperature >= 28:
            alerts.append(record["id"])
    return {"input_count": input_count, "accepted_count": len(records),
            "summary": summarize_values(values), "alerts": alerts, "issues": issues}

print(f"경고: {saved['alerts']}")

# 평균 계산은 이 예제에서 비활성 장치의 유효 측정값도 포함하는 약속입니다.
```

```python
import json

def load_rows(path):
    with path.open(encoding="utf-8", newline="") as file:
        reader = csv.DictReader(file)
        if reader.fieldnames != ["name", "quantity"]:
            raise ValueError("name, quantity 열이 필요합니다")
        return list(reader)

def inspect(rows):
    accepted = []
    issues = []
    seen = set()
    for number, row in enumerate(rows, start=2):
        # 첫 줄은 열 이름이라 실제 자료의 파일 줄은 2부터 시작합니다.

            if value is None:
                wrong_shape = True
        if wrong_shape:
            issues.append({"line": number, "reason": "열 개수 오류"})
            continue
        name = row["name"].strip()
        if name == "" or name in seen:
            issues.append({"line": number, "reason": "이름 누락 또는 중복"})
            continue
        try:
            quantity = int(row["quantity"])
            if quantity < 0:
                raise ValueError("음수 수량")
        except ValueError:
            issues.append({"line": number, "reason": "수량 오류"})
            continue
        seen.add(name)
        accepted.append({"name": name, "quantity": quantity})
    return accepted, issues

def make_report(accepted, issues):
    total = 0
    low_stock = []
    for row in accepted:
        total += row["quantity"]
        if row["quantity"] < 3:
            low_stock.append(row["name"])
    return {"accepted": len(accepted), "total": total,
            "low_stock": low_stock, "issues": issues}
```

## 첫 예제에서 사용하는 함수 파일 · sample_helpers.py

첫 예제와 같은 폴더에 두세요. 값이 미측정인지 검사하고 평균을 계산하는 함수 모음입니다. [함수 파일 받기](sample_helpers.py)

```python
# 함께 사용하는 함수 모음. sample_helpers.py로 같은 폴더에 저장합니다.
# import는 이 파일의 함수 이름을 다른 코드에서 사용할 수 있게 합니다.
# 함수 정의만 있으며 파일을 가져오는 순간 입력이나 파일 저장은 하지 않습니다.
import math

def normalize_active(raw):
    # CSV의 상태 글자를 Python 불리언으로 바꿉니다. bool(raw)를 쓰지 않습니다.
    text = raw.strip().lower()
    if text == "true":
        return True
    if text == "false":
        return False
    raise ValueError("true 또는 false가 필요합니다")

def summarize_values(values):
    total = 0
    count = 0
    missing = 0
    for value in values:
        if value is None:
            missing += 1
            continue
        # True/False, 문자열, 무한대와 NaN은 평균에 넣지 않습니다.
        # math.isfinite는 유한한 숫자인지 검사합니다.
        if type(value) not in (int, float) or not math.isfinite(value):
            raise ValueError("유한한 숫자 또는 None이 필요합니다")
        total += value
        count += 1
    average = None if count == 0 else round(total / count, 2)
    return {"count": count, "missing": missing, "average": average}
```
